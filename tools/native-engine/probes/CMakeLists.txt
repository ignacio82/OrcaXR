cmake_minimum_required(VERSION 3.22)
project(OrcaXRNativeEngineProbes LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI MATCHES "^(arm64-v8a|x86_64)$")
    message(FATAL_ERROR "This probe must use an initial Android ABI")
endif()
find_package(TBB 2021.5 REQUIRED COMPONENTS tbb tbbmalloc)
add_executable(orcaxr_tbb_probe tbb_probe.cpp)
target_compile_features(orcaxr_tbb_probe PRIVATE cxx_std_17)
target_compile_options(orcaxr_tbb_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(orcaxr_tbb_probe PRIVATE TBB::tbb TBB::tbbmalloc)
